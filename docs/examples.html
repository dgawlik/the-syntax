PR code review

Elicits a path, reads the file, loops over review aspects with for as an expression, then summarizes. Source: r01-pr-code-review.spl.

@{SPL/1}

Review a source file the way a senior engineer would in a pull request.

$>
let $file = elicit{sch='string'} 'Which file should I review? (relative path)'
try:
    $code = read{path=$file}
on_error:
    fail 'Cannot review: $file could not be read'
assert len($code) > 0 -> 'File is empty, nothing to review'

forbid 'Do not rewrite the whole file; only quote the lines that matter'

let $checks = ['correctness and edge cases', 'error handling', 'security', 'readability', 'test coverage']
let $findings = for $c in $checks:
    let $f = out{len='one or two sentences', fmt='sentence'} 'Review the code below for $c. If there is nothing to report, say "No issues". Code: $code'
    yield $f

think{level='low'} 'Rank the findings by severity and drop the ones that say "No issues"'
out{len='under 8 bullets', fmt='bullet_list'} 'Summarize the review of $file from these findings, most severe first: $findings'
<$

Commit message

Hard constraints, a tool call, a semantic condition combined with a deterministic one. Source: r02-commit-message.spl.

@{SPL/1}

Write a Conventional Commits message for what is currently staged.

$>
enforce 'Subject line is at most 72 characters, imperative mood, no trailing period'
enforce 'Subject starts with one of: feat, fix, docs, refactor, test, chore'

$diff = execute{command='git --no-pager diff --staged'} 'Get the staged changes'
assert len($diff) > 0 -> 'Nothing is staged'

let $subject = out{len='one line', fmt='sentence'} 'Write the commit subject for this diff: $diff'
if len($diff) > 2000 | ?'the change touches several unrelated concerns':
    let $body = out{len='under 5 bullets', fmt='bullet_list'} 'Explain what changed and why, grouped by concern, for this diff: $diff'
    out 'Subject: $subject Body: $body'
else:
    out 'Subject: $subject'
<$

Changelog as JSON

Structured output with a JSON schema. Because it is structured, it is the only output of the message. Source: r03-changelog-json.spl.

@{SPL/1}

Turn recent git history into machine-readable release notes.

$>
let $range = elicit{sch='string'} 'Which git range should the changelog cover? (e.g. v1.2.0..HEAD)'
$log = execute{command='git --no-pager log --oneline $range'} 'List commits in the range'
assert len($log) > 0 -> 'No commits found in range $range'

out{sch={'type': 'object', 'required': ['range', 'features', 'fixes', 'other'], 'properties': {'range': {'type': 'string'}, 'features': {'type': 'array', 'items': {'type': 'string'}}, 'fixes': {'type': 'array', 'items': {'type': 'string'}}, 'other': {'type': 'array', 'items': {'type': 'string'}}}}} 'Classify each commit below into features, fixes or other, rewriting each as a user-facing sentence. Use "$range" as the range value. Commits: $log'
<$

Release gate

Hard gates abort, soft gates warn. Combines elicit, membership tests, conditional constraints, try/on_error and negated semantic assertions. Source: r08-release-gate.spl.

@{SPL/1}

Decide whether a release may ship. Hard gates abort; soft gates produce warnings.

$>
let $version = elicit{sch='string'} 'Which version are you releasing?'
let $env = elicit{sch='string'} 'Target environment (staging or prod)?'
assert $env in ['staging', 'prod'] -> 'Unknown environment: $env'

if $env == 'prod':
    enforce 'Require a second reviewer sign-off before the release proceeds'
    forbid 'Do not recommend skipping the staging soak period'

try:
    $changelog = read{path='CHANGELOG.md'}
on_error:
    fail 'CHANGELOG.md is missing; cannot release $version'

execute{command='npm test --silent'} 'Run the full test suite'
assert !?'the test output contains failures' -> 'Tests are failing; release blocked'
assert ?'the changelog has an entry for version $version' -> 'CHANGELOG.md has no entry for the requested version'

let $warnings = out{sch='array'} 'List up to 3 risks from the changelog that deserve a manual check before shipping, as a JSON array of strings (empty if none). Changelog: $changelog'

if len($warnings) == 0:
    out 'Release $version is clear to ship to $env.'
else:
    out{fmt='bullet_list'} 'Release $version may ship to $env after these manual checks: $warnings'
<$
